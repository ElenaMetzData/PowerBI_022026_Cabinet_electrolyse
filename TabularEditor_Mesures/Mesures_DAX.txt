- Nombre de clients actifs projetés sur les 6 prochains mois (sont pris en compte : les rdv planifiés, la taux d'abandon et les nouveaux clients, et la saisonnalité)

```
Clients actifs projetés =


VAR Pivot = [Date Pivot (mois terminé)]
VAR d = MAX(Calendrier[Date])
VAR N = MAX(0, DATEDIFF(EOMONTH(Pivot, 0), EOMONTH(d, 0), MONTH))

VAR ActifsNow =
    CALCULATE(
        DISTINCTCOUNT(planity_rdv_36_mois[client_id]),
        planity_rdv_36_mois[statut] IN {"Réalisé", "Honoré"},
        REMOVEFILTERS(Calendrier),
        DATESBETWEEN(
            Calendrier[Date],
            EDATE(Pivot, -3),
            Pivot
        )
    )

VAR NewAvg = [Nouveaux clients - moy / mois (12M)]
VAR Churn = [Taux abandon - moy mensuel (12M)]
VAR ChurnSafe = MAX(0.001, MIN(Churn, 0.999))
VAR Ret = 1 - ChurnSafe
VAR NSafe = MIN(N, 24)

RETURN
IF(
    ISBLANK(ActifsNow) || ActifsNow = 0,
    BLANK(),
    ActifsNow * POWER(Ret, NSafe) + NewAvg * DIVIDE(1 - POWER(Ret, NSafe), ChurnSafe)
)
```

- Durée moyenne de suivi d’un client, en mois, entre son premier et son dernier rendez-vous réalisé, en ignorant les filtres de date.

```
Durée traitement moyenne (mois) (ignore filtres) =

VAR T =
    ADDCOLUMNS(
        VALUES(clients_36_mois[client_id]),
        "FirstDate",
            CALCULATE(
                MIN(planity_rdv_36_mois[date]),
                REMOVEFILTERS(Calendrier),
                planity_rdv_36_mois[statut] IN {"Réalisé","Honoré"}
            ),
        "LastDate",
            CALCULATE(
                MAX(planity_rdv_36_mois[date]),
                REMOVEFILTERS(Calendrier),
                planity_rdv_36_mois[statut] IN {"Réalisé","Honoré"}
            )
    )
RETURN
AVERAGEX(
    FILTER(T, NOT(ISBLANK([FirstDate])) && NOT(ISBLANK([LastDate]))),
    DIVIDE(DATEDIFF([FirstDate], [LastDate], DAY), 30.4375)
)
```


- Coefficient multiplicateur calculé sur 24 mois d'historique. Compare les heures réalisées sur un mois donné (ex: tous les juin) à la moyenne mensuelle globale. Un coefficient > 1 indique un mois historiquement plus chargé.

```
Facteur saisonnalité (mois) =

VAR Pivot = [Date Pivot (mois terminé)]
VAR MoisNum = MAX(Calendrier[Numero du mois])

-- Fenêtre historique
VAR WindowHist =
    DATESINPERIOD(Calendrier[Date], Pivot, -24, MONTH)

-- Total heures sur les mois "du même type" (ex : tous les mars)
VAR HeuresMoisType =
    CALCULATE(
        [Heures RDV réalisés],
        WindowHist,
        FILTER(
            ALL(Calendrier),
            Calendrier[Numero du mois] = MoisNum
        )
    )

-- Nb de mois de ce type dans la fenêtre (ex : nb de mars)
VAR NbMoisType =
    CALCULATE(
        DISTINCTCOUNT(Calendrier[Annee mois]),
        WindowHist,
        FILTER(
            ALL(Calendrier),
            Calendrier[Numero du mois] = MoisNum
        )
    )

VAR MoyMoisType =
    DIVIDE(HeuresMoisType, NbMoisType)

-- Moyenne mensuelle globale sur la fenêtre
VAR HeuresTotal =
    CALCULATE([Heures RDV réalisés], WindowHist)

VAR NbMoisWindow =
    CALCULATE(
        DISTINCTCOUNT(Calendrier[Annee mois]),
        WindowHist
    )

VAR MoyMensuelleGlobale =
    DIVIDE(HeuresTotal, NbMoisWindow)

RETURN
DIVIDE(MoyMoisType, MoyMensuelleGlobale)
```

- Estimation des heures de soins pour les semaines au-delà des RDV déjà calés. Calculée à partir de la moyenne mensuelle historique des 12 derniers mois, ajustée par la saisonnalité et enrichie de la contribution des nouveaux clients. Les heures déjà planifiées sont soustraites pour éviter tout double-compte.

```
Heures RDV attendues (non planifiées) =


VAR Pivot = [Date Pivot (mois terminé)]
VAR d = MAX(Calendrier[Date])

VAR HeuresMoyMensuelle =
    CALCULATE(
        DIVIDE([Heures RDV réalisés], 12),
        DATESINPERIOD(Calendrier[Date], Pivot, -12, MONTH),
        ALL(Calendrier)
    )

VAR NewAvg = [Nouveaux clients - moy / mois (12M)]
VAR HeuresParClient = [Heures par client actif (12M)]
VAR N = MAX(0, DATEDIFF(EOMONTH(Pivot, 0), EOMONTH(d, 0), MONTH))
VAR HeuresNouveauxClients = NewAvg * HeuresParClient * N
VAR Saison = [Facteur saisonnalité (mois)]
VAR HeuresProjetees = (HeuresMoyMensuelle + HeuresNouveauxClients) * Saison
VAR HeuresCalees = [Heures RDV planifiés]

RETURN
MAX(0, HeuresProjetees - HeuresCalees)
```

- Calcul de l'évolution mensuelle du taux d'abandon au cours de l'année dernière glissante

```
Taux abandon - moy mensuel (12M) =


VAR Pivot = [Date Pivot (mois terminé)]

VAR MoisHisto =
    CALCULATETABLE(
        VALUES(Calendrier[Annee mois]),
        DATESINPERIOD(Calendrier[Date], Pivot, -12, MONTH),
        ALL(Calendrier)
    )

RETURN
AVERAGEX(
    MoisHisto,
    VAR FinM =
        CALCULATE(
            MAX(Calendrier[Date]),
            ALL(Calendrier),
            Calendrier[Annee mois] = EARLIER(Calendrier[Annee mois])
        )
    VAR DebFenetre = EDATE(FinM, -3)
    VAR DebSuivant = FinM + 1
    VAR FinSuivant = EOMONTH(FinM, 1)
    VAR ActifsM =
        CALCULATETABLE(
            VALUES(planity_rdv_36_mois[client_id]),
            planity_rdv_36_mois[statut] IN {"Réalisé", "Honoré"},
            ALL(Calendrier),
            DATESBETWEEN(Calendrier[Date], DebFenetre, FinM)
        )
    VAR ActifsM1 =
        CALCULATETABLE(
            VALUES(planity_rdv_36_mois[client_id]),
            planity_rdv_36_mois[statut] IN {"Réalisé", "Honoré"},
            ALL(Calendrier),
            DATESBETWEEN(Calendrier[Date], DebSuivant, FinSuivant)
        )
    VAR Retenus = COUNTROWS(INTERSECT(ActifsM, ActifsM1))
    VAR Total = COUNTROWS(ActifsM)
    RETURN 1 - DIVIDE(Retenus, Total)
)
```
